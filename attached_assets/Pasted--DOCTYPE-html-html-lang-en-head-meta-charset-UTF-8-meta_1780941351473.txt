<script>
(function () {
  const FALLBACK = ['../assets/workshop-frame.jpg','../assets/workshop-wingback.jpg','../assets/workshop-upholstery.jpg','../assets/workshop-restoration.jpg','../assets/workshop-team.jpg','../assets/workshop-leather.jpg','../assets/workshop-designer.jpg'];
  function getImg(p, idx) {
    if (p.imageFiles && p.imageFiles.length) { const f = p.imageFiles[0]; return f.startsWith('http') ? f : '../' + f; }
    return FALLBACK[idx % FALLBACK.length];
  }

  function renderProduct(p, idx) {
    const swatches = (p.color || []).map(c => `<div class="swatch" style="background:${c}"></div>`).join('');
    const badge = p.badge ? `<span class="product__badge${p.badgeType === 'red' ? ' product__badge--red' : ''}">${p.badge}</span>` : '';
    const sku = p.sku || '';
    return `<article class="product reveal" data-cat="${p.categoryId || ''}" data-type="${p.typeId || ''}">
      <div class="product__media">
        <img src="${getImg(p, idx)}" alt="${p.name}" loading="lazy">
        ${badge}
      </div>
      <div class="product__body">
        ${sku ? `<p class="product__sku">${sku}</p>` : ''}
        <h3 class="product__name">${p.name}</h3>
        <p class="product__desc">${p.description}</p>
        ${swatches ? `<div class="product__swatches">${swatches}</div>` : ''}
        <div class="product__footer">
          <span class="product__price">Quote on request</span>
          <a href="#" class="product__more">Details →</a>
        </div>
      </div>
    </article>`;
  }

  function groupByCategory(products, categories) {
    const catMap = {};
    categories.forEach(c => { catMap[c.id] = { ...c, items: [] }; });
    const uncategorised = { id: null, name: 'Other', items: [] };
    products.forEach(p => {
      if (p.categoryId && catMap[p.categoryId]) catMap[p.categoryId].items.push(p);
      else uncategorised.items.push(p);
    });
    const result = categories.map(c => catMap[c.id]).filter(c => c.items.length > 0);
    if (uncategorised.items.length) result.push(uncategorised);
    return result;
  }

  function buildCategoryNav(groups) {
    const nav = document.getElementById('cat-nav');
    nav.innerHTML = `<a href="#all-products" class="active">All <span class="side__count">${groups.reduce((s,g)=>s+g.items.length,0)}</span></a>`;
    groups.forEach(g => {
      const a = document.createElement('a');
      a.href = `#cat-${g.id || 'other'}`;
      a.textContent = g.name;
      const cnt = document.createElement('span');
      cnt.className = 'side__count';
      cnt.textContent = g.items.length;
      a.appendChild(cnt);
      nav.appendChild(a);
    });
  }

  function buildTypeFilter(types) {
    if (!types || !types.length) return;
    const wrap = document.getElementById('type-filter-wrap');
    const bar = document.getElementById('type-filter');
    types.forEach(t => {
      const btn = document.createElement('button');
      btn.className = 'filter-btn';
      btn.dataset.typeFilter = t.id;
      btn.textContent = t.name;
      bar.appendChild(btn);
    });
    wrap.style.display = '';

    bar.addEventListener('click', e => {
      const btn = e.target.closest('.filter-btn');
      if (!btn) return;
      bar.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      const filter = btn.dataset.typeFilter;
      document.querySelectorAll('#all-products .product').forEach(card => {
        const show = filter === 'all' || card.dataset.type === filter;
        card.style.opacity = show ? '1' : '0.18';
        card.style.pointerEvents = show ? '' : 'none';
        card.style.transition = 'opacity 0.35s ease';
      });
    });
  }

  function renderGroups(groups) {
    const container = document.getElementById('all-products');
    container.innerHTML = '';
    groups.forEach((g, gi) => {
      const sec = document.createElement('div');
      sec.className = 'cat-section';
      sec.id = `cat-${g.id || 'other'}`;
      sec.innerHTML = `<div class="cat-section__head"><h2>${g.name}</h2><span class="label">${g.items.length} piece${g.items.length!==1?'s':''}</span></div>`;
      const grid = document.createElement('div');
      grid.className = 'product-grid';
      grid.innerHTML = g.items.map((p, i) => renderProduct(p, gi * 10 + i)).join('');
      sec.appendChild(grid);

      if (gi === 0 && groups.length > 1) {
        const cta = document.createElement('div');
        cta.className = 'inline-cta reveal';
        cta.innerHTML = `<div><p class="label" style="color:var(--c-red)">Don't see it?</p><p class="inline-cta__title mt-1">Custom briefs are <em>welcome.</em></p></div><a href="request.html" class="btn btn--primary">Send a brief</a>`;
        sec.appendChild(cta);
      }

      container.appendChild(sec);
    });

    document.querySelectorAll('#all-products .reveal').forEach(el => {
      const io = new IntersectionObserver(entries => { entries.forEach(e => { if (e.isIntersecting) { e.target.classList.add('visible'); io.unobserve(e.target); } }); }, { threshold: 0.08 });
      io.observe(el);
    });
  }

  async function load() {
    try {
      const [products, categories, types] = await Promise.all([
        fetch('/api/product').then(r => r.json()),
        fetch('/api/category').then(r => r.json()),
        fetch('/api/type').then(r => r.json()),
      ]);
      const groups = groupByCategory(products, categories);
      buildCategoryNav(groups);
      buildTypeFilter(types);
      renderGroups(groups);
    } catch {
      document.getElementById('product-grid').style.display = 'none';
      document.getElementById('product-error').style.display = '';
    }
  }

  document.addEventListener('DOMContentLoaded', load);
})();
</script>
